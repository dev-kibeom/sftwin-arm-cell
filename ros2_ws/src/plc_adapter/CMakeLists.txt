cmake_minimum_required(VERSION 3.8)
project(arm_cell_plc_adapter)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/plc_task_adapter DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config codesys DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(plc_receipts tests/test_receipts.py)
  ament_add_pytest_test(plc_action_future tests/test_action_future.py)
  ament_add_pytest_test(plc_modbus_action tests/test_modbus_action.py TIMEOUT 60)
endif()
ament_package()
