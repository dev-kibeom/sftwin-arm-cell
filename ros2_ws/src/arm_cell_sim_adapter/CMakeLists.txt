cmake_minimum_required(VERSION 3.8)
project(arm_cell_sim_adapter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(urdf REQUIRED)
add_library(canonical_joint_state src/canonical_joint_state.cpp)
target_compile_features(canonical_joint_state PUBLIC cxx_std_17)
target_include_directories(canonical_joint_state PUBLIC include)
ament_target_dependencies(canonical_joint_state sensor_msgs)
add_executable(joint_state_adapter src/joint_state_adapter_node.cpp)
target_link_libraries(joint_state_adapter canonical_joint_state)
ament_target_dependencies(joint_state_adapter rclcpp rosgraph_msgs urdf)
target_compile_options(joint_state_adapter PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS joint_state_adapter DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_canonical_joint_state tests/test_canonical_joint_state.cpp)
  target_link_libraries(test_canonical_joint_state canonical_joint_state)
  find_package(launch_testing_ament_cmake REQUIRED)
  # Each destructive rollback scenario must start with its own adapter process.
  add_launch_test(tests/test_joint_state_adapter_launch.py TIMEOUT 40)
  add_launch_test(tests/test_joint_state_clock_launch.py TIMEOUT 40)
endif()
ament_package()
